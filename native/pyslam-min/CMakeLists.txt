cmake_minimum_required(VERSION 3.18)
project(godeyes_pyslam_min LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
set(PYBIND11_FINDPYTHON ON)
find_package(pybind11 CONFIG REQUIRED)
find_package(Eigen3 3.3 CONFIG REQUIRED)
if(NOT EXISTS "${OpenCV_DIR}/OpenCVConfig.cmake")
  message(FATAL_ERROR "Set OpenCV_DIR to the opencv@4 prefix/lib/cmake/opencv4 directory")
endif()
find_package(OpenCV 4 REQUIRED CONFIG COMPONENTS core features2d imgproc calib3d
  PATHS "${OpenCV_DIR}" NO_DEFAULT_PATH)
if(NOT DEFINED PYSLAM_SOURCE)
  message(FATAL_ERROR "PYSLAM_SOURCE is required")
endif()
set(src "${PYSLAM_SOURCE}/cpp")
include_directories("${src}/casters" ${OpenCV_INCLUDE_DIRS})
pybind11_add_module(pyslam_utils NO_EXTRAS "${src}/utils/utils_module.cpp" "${src}/utils/utils.cpp")
pybind11_add_module(hamming NO_EXTRAS "${src}/hamming/hamming_module.cpp")
pybind11_add_module(pnpsolver NO_EXTRAS "${src}/solvers/pnpsolver_module.cpp"
  "${src}/solvers/PnPsolver.cpp" "${src}/solvers/MLPnPsolver.cpp" "${src}/solvers/Random.cpp")
# No window/camera modules: input arrives over the network and UI is in the browser.
target_link_libraries(pyslam_utils PRIVATE Eigen3::Eigen opencv_core opencv_imgproc opencv_features2d)
target_link_libraries(pnpsolver PRIVATE Eigen3::Eigen opencv_core opencv_calib3d)
target_link_libraries(hamming PRIVATE Eigen3::Eigen)
